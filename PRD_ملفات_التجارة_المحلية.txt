١. نظرة عامة على الوحدة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

الاسم الرسمي  : ملفات (التجار والعملاء)
المسار في القائمة: التجارة المحلية ← الملفات
المسار في النظام: /local-trade/parties
الوصف         :
    وحدة الملفات هي العمود الفقري لنظام التجارة المحلية. تتيح إنشاء سجل
    مركزي لكل تاجر أو عميل تتعامل معه الشركة، وتجميع جميع تعاملاته
    (فواتير، مدفوعات، هوامش، تحصيلات، كشف حساب) في ملف واحد متكامل.

الهدف الرئيسي:
    - تتبع كل طرف تجاري (تاجر / عميل / مزدوج) بسجل كامل ومستقل.
    - معرفة الرصيد المستحق بدقة في أي وقت.
    - إدارة دورة الحياة الكاملة: من الفاتورة حتى التحصيل والتسوية.

٢. صفحة قائمة الملفات — /local-trade/parties
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

٢.١ وصف الشاشة
────────────────
    شاشة رئيسية تعرض جميع الملفات المضافة في النظام على شكل جدول، مع
    أدوات بحث وفلترة وإضافة ملف جديد.

٢.٢ أعمدة الجدول
──────────────────
    ┌─────────────────┬────────────────────────────────────────────────────┐
    │ العمود          │ الوصف                                              │
    ├─────────────────┼────────────────────────────────────────────────────┤
    │ صورة            │ صورة الملف (Avatar) أو أيقونة افتراضية حسب النوع  │
    │ الاسم           │ الاسم الكامل للطرف التجاري                         │
    │ النوع           │ تاجر / عميل / مزدوج (Badge ملون)                  │
    │ المحل           │ اسم المحل أو الشركة إن وُجد                        │
    │ الهاتف          │ رقم الهاتف مع أيقونة                               │
    │ نوع الدفع       │ كاش / آجل (Badge)                                  │
    │ حد الائتمان     │ قيمة الحد أو "غير محدود" (للآجل فقط)               │
    │ الرصيد الحالي   │ الرصيد المحسوب من دفتر القيود (أخضر = دائن ، أحمر = مدين) │
    │ الحالة          │ نشط / غير نشط (Badge)                              │
    │ إجراءات         │ زر تعديل (قلم) + زر عرض الملف (عين)               │
    └─────────────────┴────────────────────────────────────────────────────┘

٢.٣ أدوات البحث والفلترة
──────────────────────────
    أ) شريط البحث النصي:
       - البحث يعمل على: الاسم، اسم المحل، ورقم الهاتف.
       - البحث فوري (real-time) بدون ضغط زر.

    ب) تبويبات الفلترة حسب النوع:
       - الكل | تاجر | عميل | مزدوج
       - يمكن تمرير نوع الفلتر عبر رابط URL (مثال: ?type=merchant)
         وهذا مفيد للوصول المباشر من صفحات أخرى.

    ج) خيار "النشطين فقط":
       - Checkbox لإخفاء الملفات غير النشطة من القائمة.

٢.٤ إضافة ملف جديد
────────────────────
    زر "إضافة ملف جديد" يفتح نموذجاً في نافذة منبثقة (Dialog) يحتوي على
    الحقول التالية:

    ┌───────────────────────┬──────────┬──────────────────────────────────┐
    │ الحقل                 │ إلزامي؟  │ الوصف                            │
    ├───────────────────────┼──────────┼──────────────────────────────────┤
    │ نوع الملف             │ نعم      │ تاجر / عميل / مزدوج              │
    │ الاسم                 │ نعم      │ الاسم الكامل                     │
    │ اسم المحل             │ لا       │ اسم المحل أو الشركة              │
    │ الهاتف                │ لا       │ رقم الهاتف                       │
    │ واتساب                │ لا       │ رقم واتساب مستقل                 │
    │ المنطقة               │ لا       │ المنطقة الجغرافية                │
    │ المحافظة              │ لا       │ اسم المحافظة                     │
    │ نوع الدفع             │ نعم      │ كاش / آجل                        │
    │ حد الائتمان           │ مشروط    │ يظهر فقط عند اختيار "آجل"        │
    │   ← نوع الحد          │ مشروط    │ غير محدود / محدود                │
    │   ← قيمة الحد         │ مشروط    │ يظهر عند اختيار "محدود"          │
    │ الرصيد الافتتاحي      │          │                                  │
    │   ← نوع الرصيد        │ نعم      │ مدين (عليه) / دائن (له)          │
    │   ← قيمة الرصيد       │ نعم      │ رقمية بالجنيه المصري             │
    │ نشط                   │ نعم      │ Switch تفعيل / تعطيل             │
    └───────────────────────┴──────────┴──────────────────────────────────┘

    ملاحظة حول الرصيد الافتتاحي:
        عند حفظ الملف، يُنشئ النظام تلقائياً قيداً في دفتر القيود
        من نوع "opening_balance" بالقيمة المُدخلة، مما يجعل الرصيد
        الافتتاحي جزءاً أصيلاً من سجل القيود وليس حقلاً منفصلاً.

٢.٥ تعديل ملف موجود
─────────────────────
    زر القلم في عمود الإجراءات يفتح نفس النموذج مملوءاً ببيانات الملف
    الحالية وقابلة للتعديل.

٣. أنواع الملفات (Party Types)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    ┌────────────┬────────────┬──────────────────────────────────────────┐
    │ القيمة     │ الاسم      │ الوصف                                    │
    ├────────────┼────────────┼──────────────────────────────────────────┤
    │ merchant   │ تاجر       │ يتم الشراء منه (فواتير شراء)             │
    │ customer   │ عميل       │ يتم البيع له (فواتير بيع)                │
    │ both       │ مزدوج      │ يظهر في كلا القائمتين (شراء وبيع)       │
    └────────────┴────────────┴──────────────────────────────────────────┘

    - الملف المزدوج (both) يمكن إصدار فواتير شراء وبيع معه في نفس الوقت.
    - النوع لا يمكن تغييره بعد إنشاء الملف (لضمان سلامة القيود).

٤. صفحة الملف التفصيلية (Party Profile) — /local-trade/parties/:id
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    هذه الصفحة هي المحور الرئيسي للعمل مع أي ملف. تحتوي على:
    - رأس صفحة لاصق (Sticky Header) يظهر دائماً عند التمرير.
    - بطاقات مؤشرات الأداء (KPI Cards).
    - ٨ تبويبات تغطي كل جوانب الملف.
    - أزرار الإجراءات السريعة.

٤.١ رأس الصفحة (Sticky Header)
────────────────────────────────
    يحتوي على:
    - Breadcrumb: الملفات ← اسم الملف
    - صورة الملف (Avatar مع حرف الاسم الأول كـ fallback)
    - اسم الملف + Badge النوع (تاجر / عميل) + Badge الحالة (موقوف)
    - اسم المحل إن وُجد
    - رقم الهاتف بأيقونته

    أزرار الإجراءات السريعة (Quick Actions):
    ┌─────────────────┬────────────────────────────────────────────────┐
    │ الزرار          │ الوظيفة                                        │
    ├─────────────────┼────────────────────────────────────────────────┤
    │ 🔔 جرس الإشعارات│ يظهر فقط عند وجود تنبيهات + عداد أحمر         │
    │ تسجيل دفعة     │ يفتح نموذج تسجيل دفعة جديدة                   │
    │ فاتورة جديدة    │ ينتقل لصفحة إنشاء فاتورة مع تحديد الملف مسبقاً │
    │ تعديل           │ يفتح نموذج تعديل بيانات الملف                  │
    │ حذف (أحمر)      │ يظهر فقط للمديرين — يفتح نافذة تأكيد الحذف    │
    └─────────────────┴────────────────────────────────────────────────┘

٤.٢ بطاقات مؤشرات الأداء (KPI Cards)
───────────────────────────────────────
    ٦ بطاقات تعطي نظرة سريعة على حالة الملف:

    ┌──────────────────┬──────────────────────────────────────────────┐
    │ البطاقة          │ ما تعرضه                                     │
    ├──────────────────┼──────────────────────────────────────────────┤
    │ إجمالي الفواتير  │ مجموع قيم كل الفواتير بالجنيه               │
    │ إجمالي المدفوع   │ مجموع المدفوعات (أخضر)                       │
    │ المتبقي          │ الرصيد الحالي مع تحديد (عليه/له) بالألوان    │
    │ تحت الفحص        │ مبلغ الهوامش قيد الفحص (أصفر برتقالي)       │
    │ آخر فاتورة       │ تاريخ آخر فاتورة صدرت                        │
    │ آخر تحصيل        │ تاريخ آخر عملية تحصيل                       │
    └──────────────────┴──────────────────────────────────────────────┘

    آلية حساب الرصيد:
        الرصيد لا يُؤخذ من حقل في جدول الملفات، بل يُحسب ديناميكياً
        من دفتر القيود (partyLedgerEntries) عبر جمع كل القيود المرتبطة
        بالملف في الموسم الحالي.
        - القيد الموجب (+) = مدين (عليه)
        - القيد السالب (−) = دائن (له)

٥.١ تبويب: نظرة عامة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:

    أ) لوحة التنبيهات (إن وُجدت تنبيهات معلقة):
       - بطاقة صفراء لتنبيهات التحصيل القادمة
       - بطاقة حمراء لتنبيهات التحصيل المتأخرة (overdue)
       - زر إغلاق لكل تنبيه (dismiss)

    ب) بيانات الملف التفصيلية:
       - النوع، الاسم، اسم المحل
       - الهاتف، واتساب
       - المنطقة، المحافظة
       - نوع الدفع (كاش/آجل) + حد الائتمان
       - الرصيد الافتتاحي
       - حالة التفعيل

    ج) ملخص مالي للموسم الحالي:
       - إجمالي الفواتير
       - إجمالي المدفوعات
       - الرصيد المتبقي مع اتجاهه (مدين/دائن)

    د) زر "تسوية الموسم":
       - يسمح بإغلاق الموسم الحالي وترحيل الرصيد للموسم الجديد
       - يشترط وجود رصيد صفري لإتمام التسوية

٥.٢ تبويب: الفواتير
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    جدول بكل الفواتير المرتبطة بالملف.

الأعمدة:
    ┌──────────────┬────────────────────────────────────────────────────┐
    │ العمود       │ الوصف                                              │
    ├──────────────┼────────────────────────────────────────────────────┤
    │ الرقم المرجعي│ رقم الفاتورة التلقائي (مثال: 10001-02052026)       │
    │ التاريخ      │ تاريخ الفاتورة                                     │
    │ النوع        │ شراء / بيع / تسوية / مرتجع                        │
    │ الحالة       │ مسودة / منشورة / مستلمة / مؤرشفة                  │
    │ الإجمالي     │ إجمالي قيمة الفاتورة بالجنيه                      │
    │ المدفوع      │ المبلغ المدفوع منها                                │
    │ المتبقي      │ الفرق (إجمالي − مدفوع)                             │
    │ حالة السداد  │ مسدد / جزئي / غير مسدد                            │
    └──────────────┴────────────────────────────────────────────────────┘

الفلاتر المتاحة:
    - فلتر حالة الفاتورة (الكل / مسودة / منشورة / مستلمة / مؤرشفة)
    - فلتر نوع الفاتورة (الكل / شراء / بيع)
    - زر "فاتورة جديدة" ينقل لصفحة إنشاء الفاتورة مع تحديد الملف

أنواع الفواتير:
    ┌──────────────┬─────────────────────────────────────────────────────┐
    │ النوع        │ التأثير على الرصيد                                  │
    ├──────────────┼─────────────────────────────────────────────────────┤
    │ شراء (purchase)│ يُضيف للمخزون، يرفع رصيد الملف (مدين للتاجر)   │
    │ بيع (sale)   │ يُنقص من المخزون، يرفع رصيد العميل (مدين لنا)      │
    │ تسوية (settlement)│ يصفّر الرصيد ويؤرشف الموسم                  │
    │ مرتجع (return)│ عكس تأثير الفاتورة الأصلية                       │
    └──────────────┴─────────────────────────────────────────────────────┘

دورة حياة الفاتورة:
    مسودة (draft) ← منشورة (posted) ← مستلمة (received) ← مؤرشفة (archived)

    - "مستلمة" تعني إدخال الكميات الفعلية المستلمة لكل سطر
    - عند الاستلام: فقط الكميات الفعلية تؤثر على المخزون
    - أي نقص بين المطلوب والمستلم يُنشئ حالة هامش تلقائياً

٥.٣ تبويب: المدفوعات
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    جدول بكل المدفوعات المسجلة مع الملف.

الأعمدة:
    ┌──────────────────┬──────────────────────────────────────────────┐
    │ العمود           │ الوصف                                        │
    ├──────────────────┼──────────────────────────────────────────────┤
    │ التاريخ          │ تاريخ الدفعة                                 │
    │ المبلغ           │ قيمة الدفعة بالجنيه                          │
    │ طريقة الدفع      │ نقدي / فودافون كاش / إنستاباي / تحويل بنكي  │
    │ المستلم          │ اسم من استلم الدفعة                          │
    │ رقم المرجع       │ رقم المرجع للمدفوعات الإلكترونية             │
    │ الفاتورة المرتبطة│ رقم الفاتورة إن كانت الدفعة مرتبطة بفاتورة  │
    │ ملاحظات         │ أي ملاحظات إضافية                             │
    └──────────────────┴──────────────────────────────────────────────┘

تسجيل دفعة جديدة:
    الحقول المطلوبة في نموذج الدفعة:
    - التاريخ (مطلوب)
    - المبلغ بالجنيه (مطلوب)
    - طريقة الدفع: نقدي، فودافون كاش، إنستاباي، تحويل بنكي، أخرى
    - اسم المستلم
    - رقم المرجع (للمدفوعات الإلكترونية)
    - ربط الدفعة بفاتورة محددة (اختياري — يُظهر قائمة الفواتير غير المسددة)
    - ملاحظات
    - مرفق (صورة أو ملف)

    آلية ربط الدفعة بفاتورة:
        عند اختيار فاتورة من قائمة الفواتير غير المسددة، يتم تعبئة
        المبلغ تلقائياً بالمبلغ المتبقي من الفاتورة. وبعد الحفظ، تتحدث
        حالة الفاتورة إلى "مسدد" أو "جزئي" حسب المبلغ.

٥.٤ تبويب: الهوامش
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    جدول بحالات الهامش والنواقص المرتبطة بالملف.

ما هو الهامش؟
    عند استلام فاتورة بكمية أقل من المطلوب في أي سطر، يُنشئ النظام
    تلقائياً "حالة هامش" للفرق. تُتابع هذه الحالة حتى يتم حلها.

الأعمدة:
    ┌──────────────┬──────────────────────────────────────────────────┐
    │ العمود       │ الوصف                                            │
    ├──────────────┼──────────────────────────────────────────────────┤
    │ الفاتورة     │ رقم الفاتورة المصدر                              │
    │ الصنف        │ الصنف الذي به نقص                                │
    │ الكميات      │ عدد الكراتين والقطع الناقصة                     │
    │ المبلغ       │ قيمة الناقص بالجنيه                              │
    │ الحالة       │ تحت الفحص / تم الحل                              │
    │ طريقة الحل   │ قبول مرتجع / استبدال / خصم / تالف / رفض         │
    └──────────────┴──────────────────────────────────────────────────┘

الفلتر:
    - الحالة: الكل / تحت الفحص / تم الحل

٥.٥ تبويب: كشف الحساب
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    دفتر القيود الكامل بكل المعاملات المالية للملف بترتيب زمني.

الأعمدة:
    ┌──────────────────┬──────────────────────────────────────────────┐
    │ العمود           │ الوصف                                        │
    ├──────────────────┼──────────────────────────────────────────────┤
    │ التاريخ          │ تاريخ القيد                                   │
    │ نوع القيد        │ فاتورة / دفعة / رصيد افتتاحي / تسوية / غيره  │
    │ البيان           │ ملاحظة توضيحية للقيد                         │
    │ المبلغ           │ القيمة موجبة (مدين) أو سالبة (دائن)           │
    │ الرصيد التراكمي  │ الرصيد بعد كل قيد                            │
    └──────────────────┴──────────────────────────────────────────────┘

أنواع القيود:
    - opening_balance : الرصيد الافتتاحي
    - invoice         : قيد فاتورة
    - payment         : قيد دفعة
    - credit_note     : إشعار دائن
    - debit_note      : إشعار مدين
    - adjustment      : تسوية يدوية
    - settlement      : تسوية موسم

ميزة التصدير:
    - تصدير كشف الحساب بصيغة PDF أو CSV للطباعة أو المشاركة
    - التصدير يدعم RTL والعربية بشكل كامل

٥.٦ تبويب: الأرشيف
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    قائمة بكل المواسم السابقة والحالية للملف.

ما هو الموسم؟
    كل ملف يعمل ضمن "موسم" يمثل دورة تجارية (مثل موسم صيف / شتاء
    أو سنة مالية). عند تسوية موسم يُغلق ويبدأ موسم جديد.

الأعمدة:
    ┌──────────────────┬──────────────────────────────────────────────┐
    │ العمود           │ الوصف                                        │
    ├──────────────────┼──────────────────────────────────────────────┤
    │ اسم الموسم       │ اسم الموسم (اختياري)                         │
    │ تاريخ البداية    │ متى بدأ الموسم                               │
    │ تاريخ الإغلاق    │ متى أُغلق (فارغ للموسم الحالي)              │
    │ الرصيد الافتتاحي │ الرصيد الذي بدأ به الموسم                   │
    │ الحالة           │ حالي / مغلق                                  │
    └──────────────────┴──────────────────────────────────────────────┘

٥.٧ تبويب: التحصيل
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    نظام مواعيد التحصيل المتعدد. لكل ملف يمكن تسجيل ٤ مواعيد تحصيل.

الحقول لكل موعد:
    ┌──────────────────┬──────────────────────────────────────────────┐
    │ الحقل            │ الوصف                                        │
    ├──────────────────┼──────────────────────────────────────────────┤
    │ تاريخ التحصيل   │ الموعد المتوقع للتحصيل                        │
    │ المبلغ المتوقع   │ المبلغ المخطط تحصيله                          │
    │ ملاحظات         │ ملاحظات إضافية                                │
    │ التذكير          │ زر إرسال تذكير (يُسجل وقت الإرسال)            │
    │ الحالة          │ معلق / تم التحصيل / مؤجل                      │
    └──────────────────┴──────────────────────────────────────────────┘

آلية التنبيهات التلقائية:
    - النظام يفحص تلقائياً مواعيد التحصيل عند فتح صفحة الملف.
    - إذا كان الموعد اليوم أو خلال ٣ أيام: تنبيه أصفر (قادم).
    - إذا كان الموعد مضى ولم يُجمع: تنبيه أحمر (متأخر/overdue).
    - التنبيهات تظهر في رأس الصفحة (أيقونة الجرس بعداد أحمر)
      وأيضاً في تبويب "نظرة عامة".

تحويل التحصيل إلى دفعة:
    عند الضغط على "تم التحصيل" في تبويب نظرة عامة، يفتح نموذج تسجيل
    دفعة جديدة مُعبأ مسبقاً بمبلغ التحصيل وتاريخه.

٥.٨ تبويب: الحركات (Timeline)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

المحتوى:
    عرض زمني كرونولوجي لكل الأنشطة المرتبطة بالملف.

الأنشطة التي تظهر في الـ Timeline:
    - إنشاء فاتورة جديدة
    - استلام فاتورة
    - تسجيل دفعة
    - إنشاء حالة هامش
    - حل حالة هامش
    - تسجيل تحصيل
    - تسوية موسم

كل عنصر يعرض:
    - التاريخ والوقت
    - نوع النشاط مع أيقونة ملونة
    - التفاصيل (رقم الفاتورة، المبلغ، إلخ)
    - اسم المستخدم الذي أجرى العملية

٦. خاصية حذف الملف (للمديرين فقط)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

الوصف:
    يمكن للمدير حذف أي ملف بشكل نهائي. الحذف كامل ويشمل جميع
    البيانات المرتبطة. العملية لا يمكن التراجع عنها.

شروط الظهور:
    - الزر لا يظهر إلا للمستخدم صاحب دور "مدير" (role = "مدير").
    - يظهر باللون الأحمر في رأس صفحة الملف.

خطوات العملية:
    ١. يضغط المدير زر "حذف" الأحمر في رأس الصفحة.
    ٢. تظهر نافذة تأكيد تذكر اسم الملف وتحذر من حذف البيانات.
    ٣. يضغط "تأكيد الحذف".
    ٤. يبدأ الحذف التسلسلي من قاعدة البيانات.
    ٥. بعد النجاح: رسالة توست "تم حذف الملف بنجاح" + انتقال لقائمة الملفات.

ترتيب الحذف من قاعدة البيانات:
    لضمان سلامة العلاقات (Foreign Keys)، الحذف يتم بهذا الترتيب الإجباري:

    المرحلة ١: party_ledger_entries (سجلات دفتر القيود)
    المرحلة ٢: local_payments (المدفوعات)
    المرحلة ٣: local_invoice_lines (بنود الفواتير)
    المرحلة ٤: local_invoices (الفواتير)
    المرحلة ٥: return_cases (حالات الهامش)
    المرحلة ٦: party_collections (مواعيد التحصيل)
    المرحلة ٧: party_seasons (المواسم)
    المرحلة ٨: parties (الملف نفسه)

    تسجيل في سجل التدقيق (Audit Log):
        بعد الحذف يُسجل النظام: مَن حذف، ماذا حذف، متى حذف.

١٠. القواعد والقيود التجارية (Business Rules)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

١٠.١ قواعد الرصيد
──────────────────
    - الرصيد يُحسب دائماً من القيود وليس من حقول الجدول.
    - الرصيد الموجب = مدين (الطرف مدين لنا).
    - الرصيد السالب = دائن (نحن مدينون للطرف).
    - الرصيد الافتتاحي يُدخل مرة واحدة عند إنشاء الملف كقيد في الدفتر.

١٠.٢ قواعد التسوية
────────────────────
    - لا يمكن تسوية موسم إلا إذا كان الرصيد صفراً.
    - عند التسوية: الموسم الحالي يُغلق وينشأ موسم جديد.
    - الرصيد المرحَّل يُسجل كـ opening_balance في الموسم الجديد.

١٠.٣ قواعد حد الائتمان
─────────────────────────
    - حد الائتمان لا يُطبق إلا على الملفات ذات نوع الدفع "آجل".
    - إذا كان الحد "محدود"، لا يُسمح بإصدار فواتير تتجاوز الحد.
    - إذا كان "غير محدود" لا يوجد أي قيد على قيمة الفواتير.

١٠.٤ قواعد الفواتير
────────────────────
    - كل فاتورة تُنشئ قيداً تلقائياً في دفتر القيود عند نشرها.
    - كل دفعة تُنشئ قيداً معاكساً في دفتر القيود عند تسجيلها.
    - الكميات الفعلية فقط (المستلمة) تؤثر على المخزون.
    - أي نقص عند الاستلام يُنشئ حالة هامش تلقائياً.

١٠.٥ قواعد الحذف
─────────────────
    - الحذف نهائي ولا يمكن التراجع عنه.
    - الحذف متاح للمديرين فقط.
    - الحذف cascade يشمل كل البيانات المرتبطة.
    - يُسجل الحذف في سجل التدقيق (Audit Log) قبل التنفيذ.

١١. تجربة المستخدم (UX) ومعايير الواجهة
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    - اللغة: عربية كاملة (RTL) في كل الصفحات والنوافذ.
    - الخطوط: Cairo و Tajawal للنصوص العربية.
    - رأس الصفحة: Sticky (يظل ظاهراً عند التمرير للأسفل).
    - التحميل: Skeleton loaders أثناء جلب البيانات.
    - الإشعارات: Toast notifications للعمليات (نجاح / فشل).
    - الألوان:
        أخضر  = أرقام دائنة / مسددة / نشطة
        أحمر  = أرقام مدينة / متأخرة / خطر
        أصفر  = قيد الفحص / تحذيرات
        رمادي = غير نشط / محايد
    - العملة: جميع المبالغ بالجنيه المصري (ج.م) مع تنسيق عربي.
    - التاريخ: بالتقويم الميلادي بتنسيق DD/MM/YYYY.
    - الشبكة: 24 عموداً في الجداول التفصيلية لمنع اقتطاع الأرقام.

وحدة الملفات توفر نظاماً متكاملاً لإدارة الأطراف التجارية يشمل:

    ✦ إدارة البيانات الأساسية (إنشاء / تعديل / حذف) مع صلاحيات محكمة
    ✦ ٣ أنواع من الملفات (تاجر / عميل / مزدوج) لمرونة كاملة
    ✦ نظام محاسبي قيود مزدوج لضمان دقة الأرصدة
    ✦ دورة فاتورة كاملة من الإصدار حتى الاستلام مع تتبع النقص
    ✦ نظام تحصيل ذكي مع تنبيهات تلقائية للمواعيد
    ✦ مواسم تجارية لتنظيم دورات العمل وأرشفتها
    ✦ كشف حساب قابل للتصدير PDF/CSV بدعم كامل للعربية
    ✦ حذف آمن بتسلسل مضبوط وسجل تدقيق لحفظ المساءلة
